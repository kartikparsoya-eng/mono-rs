```rust
use crate::filter::{Value, compare_values};
use std::collections::HashMap;

type Row = HashMap<String, Value>;
type CompoundKey = Vec<String>;

/// Checks if a parent row matches a child row on compound keys.
/// Equivalent to TS isJoinMatch(parent, parentKey, child, childKey).
/// Returns false if any key field is Value::Null on either side.
/// Returns true if parent[parentKey[i]] == child[childKey[i]] for all i.
fn is_join_match(
    parent: &Row,
    parent_key: &[String],
    child: &Row,
    child_key: &[String],
) -> bool {
    if parent_key.len() != child_key.len() {
        return false;
    }
    for (pk, ck) in parent_key.iter().zip(child_key.iter()) {
        let pv = parent.get(pk).unwrap_or(&Value::Null);
        let cv = child.get(ck).unwrap_or(&Value::Null);
        if matches!(pv, Value::Null) || matches!(cv, Value::Null) {
            return false;
        }
        if compare_values(pv, cv) != std::cmp::Ordering::Equal {
            return false;
        }
    }
    true
}

/// Builds a join constraint object from a source row.
/// Equivalent to TS buildJoinConstraint(sourceRow, sourceKey, targetKey).
/// Returns None if any sourceKey field is null (FK is null → no match possible).
/// Returns Some(HashMap) mapping targetKey[i] → sourceRow[sourceKey[i]].
fn build_join_constraint(
    source_row: &Row,
    source_key: &[String],
    target_key: &[String],
) -> Option<Row> {
    if source_key.len() != target_key.len() {
        return None;
    }
    let mut constraint = HashMap::new();
    for (sk, tk) in source_key.iter().zip(target_key.iter()) {
        let val = source_row.get(sk).unwrap_or(&Value::Null);
        if matches!(val, Value::Null) {
            return None;
        }
        constraint.insert(tk.clone(), val.clone());
    }
    Some(constraint)
}

/// Checks if two rows are equal on a compound key.
/// Equivalent to TS rowEqualsForCompoundKey(a, b, key).
fn row_equals_for_compound_key(
    a: &Row,
    b: &Row,
    key: &[String],
) -> bool {
    for k in key {
        let av = a.get(k).unwrap_or(&Value::Null);
        let bv = b.get(k).unwrap_or(&Value::Null);
        if compare_values(av, bv) != std::cmp::Ordering::Equal {
            return false;
        }
    }
    true
}
```

### 2b. JSON row parsing helper

```rust
/// Parse a JSON string into a Row (HashMap<String, Value>).
fn parse_row(json: &serde_json::Value) -> Row {
    let mut row = HashMap::new();
    if let Some(obj) = json.as_object() {
        for (k, v) in obj {
            row.insert(k.clone(), Value::from_json(v));
        }
    }
    row
}

/// Serialize a Row back to a serde_json::Value (object).
fn row_to_json(row: &Row) -> serde_json::Value {
    let mut map = serde_json::Map::new();
    for (k, v) in row {
        map.insert(k.clone(), match v {
            Value::Null => serde_json::Value::Null,
            Value::Bool(b) => serde_json::Value::Bool(*b),
            Value::Number(n) => serde_json::json!(*n),
            Value::String(s) => serde_json::Value::String(s.clone()),
        });
    }
    serde_json::Value::Object(map)
}
```

```rust
use napi_derive::napi;

/// Build a join constraint from a source row.
/// Returns constraint JSON string, or null if any FK field is null.
///
/// source_row_json: JSON string of the source row object
/// source_key: array of field names on the source side
/// target_key: array of field names on the target side
#[napi]
pub fn rust_build_join_constraint(
    source_row_json: String,
    source_key: Vec<String>,
    target_key: Vec<String>,
) -> Option<String> {
    let json: serde_json::Value = serde_json::from_str(&source_row_json).ok()?;
    let row = parse_row(&json);
    let constraint = build_join_constraint(&row, &source_key, &target_key)?;
    Some(serde_json::to_string(&row_to_json(&constraint)).unwrap())
}

/// Check if a parent row matches a child row on compound keys.
#[napi]
pub fn rust_is_join_match(
    parent_row_json: String,
    parent_key: Vec<String>,
    child_row_json: String,
    child_key: Vec<String>,
) -> bool {
    let pj: serde_json::Value = match serde_json::from_str(&parent_row_json) {
        Ok(v) => v,
        Err(_) => return false,
    };
    let cj: serde_json::Value = match serde_json::from_str(&child_row_json) {
        Ok(v) => v,
        Err(_) => return false,
    };
    is_join_match(&parse_row(&pj), &parent_key, &parse_row(&cj), &child_key)
}

/// Check if two rows are equal on a compound key.
#[napi]
pub fn rust_row_equals_for_compound_key(
    a_json: String,
    b_json: String,
    key: Vec<String>,
) -> bool {
    let aj: serde_json::Value = match serde_json::from_str(&a_json) {
        Ok(v) => v,
        Err(_) => return false,
    };
    let bj: serde_json::Value = match serde_json::from_str(&b_json) {
        Ok(v) => v,
        Err(_) => return false,
    };
    row_equals_for_compound_key(&parse_row(&aj), &parse_row(&bj), &key)
}
```

```rust
/// Batch function for the pushChildChange hot path.
/// Replaces the per-row TS loop with a single napi call.
///
/// Given a child row and an array of candidate parent rows (from fetch()),
/// this function:
/// 1. Builds the join constraint from the child row (returns null constraint if FK is null)
/// 2. Tests each parent row against the child row using isJoinMatch
/// 3. Returns which parents matched (as indices)
///
/// This avoids N napi round-trips for N parent rows.
///
/// Returns JSON: { "constraint": {...} | null, "matchResults": [true, false, ...] }
#[napi]
pub fn rust_join_push_child_batch(
    child_row_json: String,
    child_key: Vec<String>,
    parent_key: Vec<String>,
    parent_rows_json: Vec<String>,
) -> String {
    let child_json: serde_json::Value = match serde_json::from_str(&child_row_json) {
        Ok(v) => v,
        Err(_) => return r#"{"constraint":null,"matchResults":[]}"#.to_string(),
    };
    let child_row = parse_row(&child_json);

    // Step 1: build constraint
    let constraint = build_join_constraint(&child_row, &child_key, &parent_key);
    let constraint_json = match &constraint {
        Some(c) => row_to_json(c),
        None => serde_json::Value::Null,
    };

    // Step 2: batch isJoinMatch for all parent rows
    let match_results: Vec<bool> = parent_rows_json.iter().map(|pj| {
        match serde_json::from_str::<serde_json::Value>(pj) {
            Ok(v) => is_join_match(&parse_row(&v), &parent_key, &child_row, &child_key),
            Err(_) => false,
        }
    }).collect();

    // Return combined result
    let result = serde_json::json!({
        "constraint": constraint_json,
        "matchResults": match_results,
    });
    result.to_string()
}
```

**Performance note:** The JSON string approach is used here for simplicity and correctness. The key win is eliminating N napi boundary crossings — the JSON parse cost is O(fields) per row and happens in Rust, which is fast. A future optimization could use `JsObject` direct access to avoid serialization entirely, but this is the correct first step.
