```typescript
import type {Row} from '../../../zero-protocol/src/data.ts';
import type {Change} from './change.ts';
import {ChangeType} from './change-type.ts';
import {makeAddChange, makeRemoveChange} from './change.ts';
import {type Node} from './data.ts';
import {
  throwFilterOutput,
  type FilterInput,
  type FilterOperator,
  type FilterOutput,
} from './filter-operators.ts';
import type {SourceSchema} from './schema.ts';
import {RustFilterPredicate} from 'zqlite-rs';

/**
 * Rust-backed Filter operator. Delegates predicate evaluation to native code.
 * Falls back to TS Filter if predicate is not Rust-eligible (checked at construction time).
 */
export class RustFilter implements FilterOperator {
  readonly #input: FilterInput;
  readonly #rustPredicate: RustFilterPredicate;
  readonly #jsPredicate: (row: Row) => boolean; // kept for filter() method (fetch path)
  #output: FilterOutput = throwFilterOutput;

  constructor(
    input: FilterInput,
    predicate: (row: Row) => boolean,
    predicateJson: string,
  ) {
    this.#input = input;
    this.#jsPredicate = predicate;
    this.#rustPredicate = new RustFilterPredicate(predicateJson);
    input.setFilterOutput(this);
  }

  beginFilter(): void {
    this.#output.beginFilter();
  }

  endFilter(): void {
    this.#output.endFilter();
  }

  *filter(node: Node): Generator<'yield', boolean> {
    // filter() is used during fetch — keep in JS since it yields
    return this.#jsPredicate(node.row) && (yield* this.#output.filter(node));
  }

  setFilterOutput(output: FilterOutput) {
    this.#output = output;
  }

  destroy(): void {
    this.#input.destroy();
  }

  getSchema(): SourceSchema {
    return this.#input.getSchema();
  }

  *push(change: Change) {
    // Use Rust batch evaluation for the predicate check
    const result = this.#rustPredicate.filterPushBatch([change]);
    
    // Process results: passed changes forward directly, splits become add/remove
    for (const entry of result.passed) {
      yield* this.#output.push(change, this);
    }
    for (const split of result.splits) {
      if (split.type === 'remove') {
        yield* this.#output.push(
          makeRemoveChange(change[2] as Node), // oldNode for edit
          this,
        );
      } else if (split.type === 'add') {
        yield* this.#output.push(
          makeAddChange(change[1] as Node),
          this,
        );
      } else if (split.type === 'both') {
        yield* this.#output.push(change, this);
      }
    }
  }
}
```

Note: The actual implementation will need refinement based on the exact napi return format from Plan 01. The key pattern is:
- push() sends change to Rust for predicate evaluation
- Rust returns which changes pass and which EDITs need splitting
- TS wrapper constructs the appropriate Change tuples and forwards to output
- Relationships are never sent to Rust — they stay on the Node objects in JS

Create `packages/zql/src/ivm/rust-take-storage.ts`:

```typescript
import type {JSONValue} from '../../../shared/src/json.ts';
import type {Storage} from './operator.ts';
import type {Stream} from './stream.ts';
import {RustTakeState} from 'zqlite-rs';

/**
 * Storage adapter that delegates to Rust's in-memory HashMap
 * via RustTakeState. Replaces DatabaseStorage for Take operators.
 * 
 * Performance gain: no SQLite reads/writes, no JSON.parse/stringify
 * per get/set (Rust stores structured data directly).
 */
export class RustTakeStorage implements Storage {
  readonly #state: RustTakeState;

  constructor(sortJson: string) {
    this.#state = new RustTakeState(sortJson);
  }

  set(key: string, value: JSONValue): void {
    this.#state.setState(key, JSON.stringify(value));
  }

  get(key: string, def?: JSONValue): JSONValue | undefined {
    const result = this.#state.getState(key);
    if (result === null || result === undefined) {
      return def;
    }
    return JSON.parse(result);
  }

  *scan(options?: {prefix: string}): Stream<[string, JSONValue]> {
    // Take only uses scan for cleanup — not performance critical
    // For now, fall through to a simple implementation
    // Note: RustTakeState doesn't expose iteration — this is fine
    // because Take never calls scan() during push
    throw new Error('scan not implemented for RustTakeStorage');
  }

  del(key: string): void {
    this.#state.delState(key);
  }

  /** Expose the Rust state for direct comparison operations */
  get rustState(): RustTakeState {
    return this.#state;
  }

  destroy(): void {
    this.#state.clear();
  }
}
```

Wait — looking at Take.ts more carefully, it uses `this.#storage` typed as `TakeStorage` (a narrowed interface), not the full `Storage` interface. The storage is passed in as `Storage` but cast to `TakeStorage`. The operations used are:
- `get(key)` → returns `TakeState | Row | undefined`
- `set(key, value)` → stores `TakeState | Row`
- `del(key)`

No `scan()` is ever called by Take. So the adapter only needs get/set/del.

The real perf win here is that DatabaseStorage does:
- `get`: SQLite SELECT + JSON.parse on the result
- `set`: SQLite INSERT/UPDATE + JSON.stringify on the value

RustTakeStorage does:
- `get`: napi call → Rust HashMap lookup → JSON string back
- `set`: napi call → JSON parse in Rust → HashMap insert

This is still doing JSON serialization across the boundary. For v1 of the integration this is acceptable — the bigger win comes when we can pass structured objects via Buffer protocol (future optimization).
</action>
