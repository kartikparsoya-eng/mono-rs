Rewrite `packages/zqlite/src/db.ts` to wrap the Rust module:

```typescript
import {trace, type Attributes} from '@opentelemetry/api';
import type {LogContext} from '@rocicorp/logger';
import {Database as RustDatabase, type DatabaseOptions} from 'zqlite-rs';
import {manualSpan} from '../../otel/src/span.ts';
import {version} from '../../otel/src/version.ts';

const tracer = trace.getTracer('view-syncer', version);
const AUTO_VACUUM_INCREMENTAL = 2;
const MB = 1024 * 1024;

function mb(bytes: number): string {
  return (bytes / MB).toFixed(2);
}

export class Database implements Disposable {
  readonly #db: RustDatabase;
  readonly #threshold: number;
  readonly #lc: LogContext;
  readonly #pageSize: number;

  constructor(
    lc: LogContext,
    path: string,
    options?: DatabaseOptions,
    slowQueryThreshold = 100,
  ) {
    try {
      this.#lc = lc.withContext('class', 'Database').withContext('path', path);
      this.#db = new RustDatabase(path, options);
      this.#threshold = slowQueryThreshold;
      const [{page_size: pageSize}] = this.pragma<{page_size: number}>('page_size');
      this.#pageSize = pageSize;
    } catch (cause) {
      throw new DatabaseInitError(String(cause), {cause});
    }
  }

  // ... all methods follow same pattern:
  // 1. Start timer with performance.now()
  // 2. Call this.#db.method()
  // 3. On SqliteError: augment message with SQL
  // 4. In finally: call logIfSlow()
```

**Critical details:**
- `prepare(sql)` returns a TS `Statement` wrapping Rust `Statement` — NOT the raw Rust Statement
- `exec(sql)` wraps `#db.exec(sql)` with timing
- `pragma<T>(sql)` wraps `#db.pragma(sql)` — Rust returns array of objects, TS casts to `T[]`
- `compact()` — TS handles all the logging (freeable MB, compaction stats). Rust just does the pragma queries and vacuum.
- `transaction<T>(fn)` wraps `#db.transaction(fn)` — Rust handles BEGIN/COMMIT/ROLLBACK
- `close()` — calls `#db.close()` (Rust handles PRAGMA optimize)
- `unsafeMode()` — calls `#db.unsafeMode()`
- `name` getter — `#db.name`
- `inTransaction` getter — `#db.inTransaction`
- `[Symbol.dispose]()` — calls `close()`

**Error handling:** Rust returns errors as napi::Error. The TS wrapper must:
1. Check if it's a SqliteError (by message pattern or error code)
2. Append SQL to message: `e.message += ': ${sql}'`
3. Re-throw

Create a `SqliteError` class in db.ts (or re-export from `@rocicorp/zero-sqlite3` if still needed for type compatibility).
</action>

The Statement class in the rewritten `db.ts` wraps Rust Statement:

```typescript
export class Statement {
  readonly #stmt: RustStatement;
  readonly #lc: LogContext;
  readonly #threshold: number;
  readonly #attrs: Attributes;

  constructor(lc: LogContext, attrs: Attributes, stmt: RustStatement, threshold: number) {
    this.#lc = lc.withContext('class', 'Statement');
    this.#attrs = attrs;
    this.#stmt = stmt;
    this.#threshold = threshold;
  }

  safeIntegers(useBigInt: boolean): this {
    this.#stmt.safeIntegers(useBigInt);
    return this;
  }

  run(...params: unknown[]): RunResult {
    const start = performance.now();
    const ret = this.#stmt.run(...params);
    logIfSlow(this.#lc.withContext('method', 'run'), performance.now() - start, {...this.#attrs, method: 'run'}, this.#threshold);
    return ret;
  }

  get<T>(...params: unknown[]): T {
    const start = performance.now();
    const ret = this.#stmt.get(...params);
    logIfSlow(this.#lc.withContext('method', 'get'), performance.now() - start, {...this.#attrs, method: 'get'}, this.#threshold);
    return ret as T;
  }

  all<T>(...params: unknown[]): T[] {
    const start = performance.now();
    const ret = this.#stmt.all(...params);
    logIfSlow(this.#lc.withContext('method', 'all'), performance.now() - start, {...this.#attrs, method: 'all'}, this.#threshold);
    return ret as T[];
  }

  iterate<T>(...params: unknown[]): IterableIterator<T> {
    // Wrap Rust RowIterator in LoggingIterableIterator
    const rustIterator = this.#stmt.iterate(...params);
    return new LoggingIterableIterator(
      this.#lc.withContext('method', 'iterate'),
      this.#attrs,
      rustIterator,
      this.#threshold,
    ) as IterableIterator<T>;
  }
}
```

The `LoggingIterableIterator` wraps the Rust `RowIterator`:
- `next()` calls `rustIterator.next()` which returns `{value, done}`
- Tracks timing per-row
- On done: logs total + sqlite time

**scanStatusV2 and scanStatusReset:** These are better-sqlite3 specific. Options:
1. Stub them (return no-op functions) — verify no tests depend on them
2. Implement in Rust via raw SQLite `sqlite3_stmt_scanstatus_v2` API
For Phase 1: Stub them. Search codebase for usage first.
