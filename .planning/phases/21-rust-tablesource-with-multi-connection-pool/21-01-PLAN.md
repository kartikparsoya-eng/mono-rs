Add `check_exists` to `RustTableSource`:

```rust
pub fn check_exists(&self, row: &Row) -> Result<bool> {
    let where_clause: Vec<String> = self.primary_key.iter()
        .map(|k| format!("\"{}\" IS ?", k))
        .collect();
    let sql = format!(
        "SELECT 1 AS \"exists\" FROM \"{}\" WHERE {} LIMIT 1",
        self.table_name,
        where_clause.join(" AND ")
    );
    let params: Vec<Value> = self.primary_key.iter()
        .map(|k| row.get(k).cloned().unwrap_or(Value::Null))
        .collect();

    let pooled = self.pool.get()?;
    let rows = execute_query(&pooled, &sql, &params)?;
    Ok(!rows.is_empty())
}
```

Note: Uses `IS` instead of `=` for NULL-safe comparison (matching TS `checkExists` statement pattern).

```rust
pub fn write_change(&self, change: &SourceChange) -> Result<()> {
    match change {
        SourceChange::Add(row) => {
            let cols: Vec<String> = self.columns.iter()
                .map(|c| format!("\"{}\"", c))
                .collect();
            let placeholders: Vec<&str> = self.columns.iter().map(|_| "?").collect();
            let sql = format!(
                "INSERT INTO \"{}\" ({}) VALUES ({})",
                self.table_name,
                cols.join(", "),
                placeholders.join(", ")
            );
            let params: Vec<Value> = self.columns.iter()
                .map(|c| {
                    let val = row.get(c).cloned().unwrap_or(Value::Null);
                    let col_type = self.column_types.get(c).unwrap_or(&ColumnType::String);
                    crate::query_builder::to_sqlite_type(&val, col_type)
                })
                .collect();
            let mut stmt = self.write_conn.prepare(&sql)?;
            for (i, val) in params.iter().enumerate() {
                bind_json_param(&mut stmt, i + 1, val)?;
            }
            stmt.raw_execute()?;
        }
        SourceChange::Remove(row) => {
            let where_clause: Vec<String> = self.primary_key.iter()
                .map(|k| format!("\"{}\" = ?", k))
                .collect();
            let sql = format!(
                "DELETE FROM \"{}\" WHERE {}",
                self.table_name,
                where_clause.join(" AND ")
            );
            let params: Vec<Value> = self.primary_key.iter()
                .map(|c| {
                    let val = row.get(c).cloned().unwrap_or(Value::Null);
                    let col_type = self.column_types.get(c).unwrap_or(&ColumnType::String);
                    crate::query_builder::to_sqlite_type(&val, col_type)
                })
                .collect();
            let mut stmt = self.write_conn.prepare(&sql)?;
            for (i, val) in params.iter().enumerate() {
                bind_json_param(&mut stmt, i + 1, val)?;
            }
            stmt.raw_execute()?;
        }
        SourceChange::Edit { row, old_row } => {
            // Check if PK changed
            let pk_changed = self.primary_key.iter().any(|k| {
                row.get(k) != old_row.get(k)
            });

            if pk_changed || self.columns.len() == self.primary_key.len() {
                // DELETE old + INSERT new
                self.write_change(&SourceChange::Remove(old_row.clone()))?;
                self.write_change(&SourceChange::Add(row.clone()))?;
            } else {
                // UPDATE non-PK columns WHERE PK matches
                let non_pk_cols: Vec<&String> = self.columns.iter()
                    .filter(|c| !self.primary_key.contains(c))
                    .collect();
                let set_clause: Vec<String> = non_pk_cols.iter()
                    .map(|c| format!("\"{}\" = ?", c))
                    .collect();
                let where_clause: Vec<String> = self.primary_key.iter()
                    .map(|k| format!("\"{}\" = ?", k))
                    .collect();
                let sql = format!(
                    "UPDATE \"{}\" SET {} WHERE {}",
                    self.table_name,
                    set_clause.join(", "),
                    where_clause.join(" AND ")
                );

                // Merge old_row + row for complete data
                let mut merged = old_row.clone();
                for (k, v) in row.iter() {
                    merged.insert(k.clone(), v.clone());
                }

                let mut params: Vec<Value> = Vec::new();
                for c in &non_pk_cols {
                    let val = merged.get(*c).cloned().unwrap_or(Value::Null);
                    let col_type = self.column_types.get(*c).unwrap_or(&ColumnType::String);
                    params.push(crate::query_builder::to_sqlite_type(&val, col_type));
                }
                for k in &self.primary_key {
                    let val = merged.get(k).cloned().unwrap_or(Value::Null);
                    let col_type = self.column_types.get(k).unwrap_or(&ColumnType::String);
                    params.push(crate::query_builder::to_sqlite_type(&val, col_type));
                }

                let mut stmt = self.write_conn.prepare(&sql)?;
                for (i, val) in params.iter().enumerate() {
                    bind_json_param(&mut stmt, i + 1, val)?;
                }
                stmt.raw_execute()?;
            }
        }
    }
    Ok(())
}
```

#[test]
fn test_write_change_add() {
    let db = setup_test_db();
    let mut src = make_source(&db);
    let cid = src.connect(Some(vec![("id".into(), "asc".into())]), None, None);

    let mut new_row = Row::new();
    new_row.insert("id".into(), json!("4"));
    new_row.insert("name".into(), json!("Dan"));
    new_row.insert("age".into(), json!(28));

    src.write_change(&SourceChange::Add(new_row)).unwrap();

    // Refresh snapshot so pool sees the write
    src.set_snapshot().unwrap();

    let nodes = src.fetch(cid, &FetchRequest::default()).unwrap();
    assert_eq!(nodes.len(), 4);
    assert_eq!(nodes[3].row.get("name").unwrap(), &json!("Dan"));
}
