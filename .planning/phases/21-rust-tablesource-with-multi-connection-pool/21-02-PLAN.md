# Plan 21-02: gen_push() and gen_push_and_write_with_split_edit()

<objective>
Port the TS push path to Rust: gen_push() fans out a SourceChange to all connected pipelines with overlay management and epoch tracking, and gen_push_and_write_with_split_edit() handles the split-edit optimization (breaking EDIT into REMOVE+ADD when splitEditKeys change) and enforces write-after-push ordering (D-11).
</objective>

<must_haves>

- gen_push() iterates all connections with output_connected=true, sets overlay + epoch, creates Change from SourceChange, returns Vec<(usize, Change)> (connection_id, change) pairs
- gen_push_and_write_with_split_edit() checks splitEditKeys, optionally splits EDIT into REMOVE+ADD, calls gen_push then write_change (write-after-push ordering)
- Overlay is set before push and cleared after all connections are processed
- lastPushedEpoch is updated per-connection during push
- Assertions: ADD must not already exist, REMOVE must exist, EDIT old_row must exist
  </must_haves>

```rust
/// Result of pushing a change through connected pipelines.
/// Each entry is (connection_id, output_change).
pub struct PushResult {
    pub changes: Vec<(usize, Change)>,
}

impl RustTableSource {
    pub fn gen_push(
        &mut self,
        change: &SourceChange,
        push_epoch: u64,
    ) -> Result<PushResult> {
        // Validate existence
        match change {
            SourceChange::Add(row) => {
                assert!(
                    !self.check_exists(row)?,
                    "gen_push ADD: row already exists"
                );
            }
            SourceChange::Remove(row) => {
                assert!(
                    self.check_exists(row)?,
                    "gen_push REMOVE: row not found"
                );
            }
            SourceChange::Edit { old_row, .. } => {
                assert!(
                    self.check_exists(old_row)?,
                    "gen_push EDIT: old row not found"
                );
            }
        }

        let mut results = Vec::new();

        for (idx, conn) in self.connections.iter_mut().enumerate() {
            if !conn.output_connected {
                continue;
            }

            conn.last_pushed_epoch = push_epoch;

            let output_change = match change {
                SourceChange::Add(row) => Change::Add(Node {
                    row: row.clone(),
                    relationships: HashMap::new(),
                }),
                SourceChange::Remove(row) => Change::Remove(Node {
                    row: row.clone(),
                    relationships: HashMap::new(),
                }),
                SourceChange::Edit { row, old_row } => Change::Edit {
                    node: Node {
                        row: row.clone(),
                        relationships: HashMap::new(),
                    },
                    old_node: Node {
                        row: old_row.clone(),
                        relationships: HashMap::new(),
                    },
                },
            };

            results.push((idx, output_change));
        }

        Ok(PushResult { changes: results })
    }
}
```

Note: The overlay is managed by the caller (gen_push_and_write_with_split_edit). In TS, genPush sets overlay per-connection before push and clears it after. In Rust, since we return Change objects rather than yielding through an operator tree inline, we set the overlay before calling gen_push and clear it after the caller has used the results for downstream push.

However, to match TS semantics exactly (overlay must be visible during downstream fetch calls that happen during push), we set overlay BEFORE gen_push returns:

Update gen_push to set overlay at the start and clear at the end:

```rust
pub fn gen_push(
    &mut self,
    change: &SourceChange,
    push_epoch: u64,
) -> Result<PushResult> {
    // ... existence checks ...

    // Set overlay so that downstream re-fetches during push see the pending change
    self.overlay = Some(Overlay {
        epoch: push_epoch,
        change: change.clone(),
    });

    let mut results = Vec::new();
    for (idx, conn) in self.connections.iter_mut().enumerate() {
        if !conn.output_connected {
            continue;
        }
        conn.last_pushed_epoch = push_epoch;

        let output_change = match change {
            // ... same as above ...
        };
        results.push((idx, output_change));
    }

    // NOTE: overlay is NOT cleared here — it stays set until all downstream
    // operators have processed the change (they may call fetch() which needs
    // the overlay). The caller clears it after downstream processing.

    Ok(PushResult { changes: results })
}

/// Clear the overlay after downstream processing is complete.
pub fn clear_overlay(&mut self) {
    self.overlay = None;
}
```

```rust
/// Full push-and-write cycle. Handles split-edit optimization and
/// enforces write-after-push ordering (D-11).
///
/// Returns all (connection_id, Change) pairs produced by the push.
pub fn gen_push_and_write_with_split_edit(
    &mut self,
    change: &SourceChange,
) -> Result<Vec<(usize, Change)>> {
    let mut all_changes = Vec::new();

    if let SourceChange::Edit { row, old_row } = change {
        // Check if any connection's splitEditKeys have changed values
        let should_split = self.connections.iter().any(|conn| {
            if let Some(ref keys) = conn.split_edit_keys {
                keys.iter().any(|k| {
                    let old_val = old_row.get(k).unwrap_or(&Value::Null);
                    let new_val = row.get(k).unwrap_or(&Value::Null);
                    old_val != new_val
                })
            } else {
                false
            }
        });

        if should_split {
            // Split into REMOVE + ADD
            let remove = SourceChange::Remove(old_row.clone());
            let epoch1 = self.next_push_epoch();
            let result1 = self.gen_push(&remove, epoch1)?;
            all_changes.extend(result1.changes);
            self.write_change(&remove)?;
            self.clear_overlay();

            let add = SourceChange::Add(row.clone());
            let epoch2 = self.next_push_epoch();
            let result2 = self.gen_push(&add, epoch2)?;
            all_changes.extend(result2.changes);
            self.write_change(&add)?;
            self.clear_overlay();

            return Ok(all_changes);
        }
    }

    // Non-split path
    let epoch = self.next_push_epoch();
    let result = self.gen_push(change, epoch)?;
    all_changes.extend(result.changes);
    // Write AFTER push (D-11: write-after-push ordering)
    self.write_change(change)?;
    self.clear_overlay();

    Ok(all_changes)
}
```

Key invariants preserved:

1. **Write-after-push (D-11):** `write_change` is called AFTER `gen_push` returns changes
2. **Split-edit:** When splitEditKeys differ between old/new row, EDIT becomes REMOVE then ADD (two separate push+write cycles)
3. **Overlay lifecycle:** Set during gen_push, cleared after write

```rust
#[test]
fn test_gen_push_add() {
    let db = setup_test_db();
    let mut src = make_source(&db);
    let cid = src.connect(Some(vec![("id".into(), "asc".into())]), None, None);
    src.set_output(cid, true).unwrap();

    let mut new_row = Row::new();
    new_row.insert("id".into(), json!("4"));
    new_row.insert("name".into(), json!("Dan"));
    new_row.insert("age".into(), json!(28));

    let change = SourceChange::Add(new_row);
    let result = src.gen_push_and_write_with_split_edit(&change).unwrap();

    assert_eq!(result.len(), 1);
    assert_eq!(result[0].0, cid);
    // Verify the change was a Change::Add
    match &result[0].1 {
        Change::Add(node) => {
            assert_eq!(node.row.get("name").unwrap(), &json!("Dan"));
        }
        _ => panic!("expected Add change"),
    }

    // Verify write happened
    src.set_snapshot().unwrap();
    let nodes = src.fetch(cid, &FetchRequest::default()).unwrap();
    assert_eq!(nodes.len(), 4);
}

#[test]
fn test_gen_push_split_edit() {
    let db = setup_test_db();
    let mut src = make_source(&db);

    let mut split_keys = HashSet::new();
    split_keys.insert("name".into());
    let cid = src.connect(
        Some(vec![("id".into(), "asc".into())]),
        None,
        Some(split_keys),
    );
    src.set_output(cid, true).unwrap();

    let mut old_row = Row::new();
    old_row.insert("id".into(), json!("1"));
    old_row.insert("name".into(), json!("Alice"));
    old_row.insert("age".into(), json!(30));

    let mut new_row = Row::new();
    new_row.insert("id".into(), json!("1"));
    new_row.insert("name".into(), json!("Alicia")); // name changed => split
    new_row.insert("age".into(), json!(30));

    let change = SourceChange::Edit { row: new_row, old_row };
    let result = src.gen_push_and_write_with_split_edit(&change).unwrap();

    // Split edit produces 2 changes: Remove + Add
    assert_eq!(result.len(), 2);
    match &result[0].1 {
        Change::Remove(node) => {
            assert_eq!(node.row.get("name").unwrap(), &json!("Alice"));
        }
        _ => panic!("expected Remove change from split edit"),
    }
    match &result[1].1 {
        Change::Add(node) => {
            assert_eq!(node.row.get("name").unwrap(), &json!("Alicia"));
        }
        _ => panic!("expected Add change from split edit"),
    }
}
```
