```rust
use napi_derive::napi;
use napi::bindgen_prelude::*;

#[napi(js_name = "RustTableSource")]
pub struct RustTableSourceNapi {
    inner: RustTableSource,
}

#[napi]
impl RustTableSourceNapi {
    #[napi(constructor)]
    pub fn new(
        db_path: String,
        pool_size: u32,
        table_name: String,
        columns: Vec<String>,
        column_types: serde_json::Value, // JSON object: { col: "string"|"number"|"boolean"|"json" }
        primary_key: Vec<String>,
    ) -> napi::Result<Self> {
        let ct: HashMap<String, ColumnType> = if let Some(obj) = column_types.as_object() {
            obj.iter()
                .map(|(k, v)| {
                    let ct = match v.as_str().unwrap_or("string") {
                        "boolean" => ColumnType::Boolean,
                        "number" => ColumnType::Number,
                        "json" => ColumnType::Json,
                        "null" => ColumnType::Null,
                        _ => ColumnType::String,
                    };
                    (k.clone(), ct)
                })
                .collect()
        } else {
            HashMap::new()
        };

        let inner = RustTableSource::new(
            &db_path,
            pool_size as usize,
            table_name,
            columns,
            ct,
            primary_key,
        )
        .map_err(|e| napi::Error::from_reason(e.to_string()))?;

        Ok(Self { inner })
    }

    #[napi]
    pub fn connect(
        &mut self,
        ordering: Option<serde_json::Value>,
        filters: Option<serde_json::Value>,
        split_edit_keys: Option<Vec<String>>,
    ) -> napi::Result<u32> {
        let ord: Option<Ordering> = ordering.and_then(|v| {
            serde_json::from_value(v).ok()
        });
        let filt: Option<Condition> = filters.and_then(|v| {
            serde_json::from_value(v).ok()
        });
        let keys: Option<HashSet<String>> = split_edit_keys
            .map(|v| v.into_iter().collect());

        let id = self.inner.connect(ord, filt, keys);
        Ok(id as u32)
    }

    #[napi]
    pub fn set_output(&mut self, connection_id: u32, connected: bool) -> napi::Result<()> {
        self.inner
            .set_output(connection_id as usize, connected)
            .map_err(|e| napi::Error::from_reason(e.to_string()))
    }

    #[napi]
    pub fn fetch(
        &self,
        connection_id: u32,
        request: serde_json::Value,
    ) -> napi::Result<serde_json::Value> {
        let req: FetchRequest = serde_json::from_value(request)
            .map_err(|e| napi::Error::from_reason(format!("invalid FetchRequest: {e}")))?;
        let nodes = self
            .inner
            .fetch(connection_id as usize, &req)
            .map_err(|e| napi::Error::from_reason(e.to_string()))?;
        serde_json::to_value(&nodes)
            .map_err(|e| napi::Error::from_reason(e.to_string()))
    }

    #[napi]
    pub fn gen_push(
        &mut self,
        change: serde_json::Value,
    ) -> napi::Result<serde_json::Value> {
        let sc: SourceChange = deserialize_source_change(&change)
            .map_err(|e| napi::Error::from_reason(e))?;
        let results = self
            .inner
            .gen_push_and_write_with_split_edit(&sc)
            .map_err(|e| napi::Error::from_reason(e.to_string()))?;

        // Serialize as array of [connection_id, change]
        let out: Vec<serde_json::Value> = results
            .into_iter()
            .map(|(id, change)| {
                serde_json::json!([id, serde_json::to_value(&change).unwrap_or_default()])
            })
            .collect();
        Ok(serde_json::Value::Array(out))
    }

    #[napi]
    pub fn set_snapshot(&self) -> napi::Result<()> {
        self.inner
            .set_snapshot()
            .map_err(|e| napi::Error::from_reason(e.to_string()))
    }

    #[napi(getter)]
    pub fn table_name(&self) -> String {
        self.inner.table_name().to_string()
    }
}

fn deserialize_source_change(v: &serde_json::Value) -> std::result::Result<SourceChange, String> {
    // TS SourceChange is a tuple: [type, row] or [type, row, oldRow]
    // type: 0=add, 1=remove, 2=edit
    let arr = v.as_array().ok_or("SourceChange must be an array")?;
    if arr.len() < 2 {
        return Err("SourceChange must have at least 2 elements".into());
    }
    let change_type = arr[0].as_u64().ok_or("change type must be a number")?;
    let row: Row = arr[1]
        .as_object()
        .ok_or("row must be an object")?
        .clone();

    match change_type {
        0 => Ok(SourceChange::Add(row)),
        1 => Ok(SourceChange::Remove(row)),
        2 => {
            let old_row: Row = arr
                .get(2)
                .and_then(|v| v.as_object())
                .ok_or("edit change must have oldRow at index 2")?
                .clone();
            Ok(SourceChange::Edit { row, old_row })
        }
        _ => Err(format!("unknown change type: {change_type}")),
    }
}
```

```rust
#[test]
fn test_full_lifecycle_connect_fetch_push_write() {
    let db = setup_test_db();
    let mut src = make_source(&db);

    // Connect with ordering
    let c1 = src.connect(Some(vec![("id".into(), "asc".into())]), None, None);
    src.set_output(c1, true).unwrap();

    // Initial fetch
    let nodes = src.fetch(c1, &FetchRequest::default()).unwrap();
    assert_eq!(nodes.len(), 3);

    // Push an ADD
    let mut new_row = Row::new();
    new_row.insert("id".into(), json!("4"));
    new_row.insert("name".into(), json!("Dan"));
    new_row.insert("age".into(), json!(28));

    let changes = src
        .gen_push_and_write_with_split_edit(&SourceChange::Add(new_row))
        .unwrap();
    assert_eq!(changes.len(), 1);

    // Refresh snapshot and verify
    src.set_snapshot().unwrap();
    let nodes = src.fetch(c1, &FetchRequest::default()).unwrap();
    assert_eq!(nodes.len(), 4);

    // Push a REMOVE
    let mut del_row = Row::new();
    del_row.insert("id".into(), json!("2"));
    del_row.insert("name".into(), json!("Bob"));
    del_row.insert("age".into(), json!(25));

    let changes = src
        .gen_push_and_write_with_split_edit(&SourceChange::Remove(del_row))
        .unwrap();
    assert_eq!(changes.len(), 1);

    src.set_snapshot().unwrap();
    let nodes = src.fetch(c1, &FetchRequest::default()).unwrap();
    assert_eq!(nodes.len(), 3);
    // Verify Bob is gone
    assert!(nodes.iter().all(|n| n.row.get("name").unwrap() != &json!("Bob")));
}

#[test]
fn test_multiple_connections_push() {
    let db = setup_test_db();
    let mut src = make_source(&db);

    let c1 = src.connect(Some(vec![("id".into(), "asc".into())]), None, None);
    let c2 = src.connect(Some(vec![("name".into(), "asc".into())]), None, None);
    src.set_output(c1, true).unwrap();
    src.set_output(c2, true).unwrap();

    let mut new_row = Row::new();
    new_row.insert("id".into(), json!("4"));
    new_row.insert("name".into(), json!("Dan"));
    new_row.insert("age".into(), json!(28));

    let changes = src
        .gen_push_and_write_with_split_edit(&SourceChange::Add(new_row))
        .unwrap();

    // Both connections should receive the change
    assert_eq!(changes.len(), 2);
    assert_eq!(changes[0].0, c1);
    assert_eq!(changes[1].0, c2);
}
```
