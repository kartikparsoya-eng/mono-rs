3. **Add `#rustHydrate()` private method** to PipelineDriver class (after `#rustAdvance()` ~L974):

   ```typescript
   *#rustHydrate(
     queryID: string,
     resolvedQuery: AST,
     companions: readonly CompanionPipeline[],
   ): Iterable<RowChange | 'yield'> {
     assert(rustHydrateFn, 'Rust hydrate not available');
     const db = this.#snapshotter.current().db;
     const queriesJson = JSON.stringify([{
       query_id: queryID,
       ast: resolvedQuery,
       primary_key: [...(this.#primaryKeys?.get(resolvedQuery.table ?? '') ?? [])],
     }]);
     const resultBuf = rustHydrateFn(db.db.name, queriesJson);
     const decoded = decodeAdvanceResultBuf(resultBuf);
     if (decoded.error) {
       throw new Error(`Rust hydration failed: ${decoded.error}`);
     }
     for (const change of decoded.changes) {
       const type = change.type === 'add' ? ChangeType.ADD
         : change.type === 'edit' ? ChangeType.EDIT
         : ChangeType.REMOVE;
       yield {
         type,
         queryID,
         table: change.table,
         rowKey: change.row,  // row IS the key for hydration adds
         row: change.row as Row,
       } as RowChange;
     }
   }
   ```

   Import `decodeAdvanceResultBuf` from `./decode-advance-buf.ts` at top of file.
