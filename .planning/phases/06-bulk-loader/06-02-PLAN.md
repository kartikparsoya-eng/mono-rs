```rust
#[napi]
impl Database {
    /// Bulk INSERT: single napi crossing for entire flush buffer.
    /// sql: the batch INSERT statement (e.g., INSERT INTO t VALUES (?,?,...) × batch_size)
    /// flat_values: flat array of all values [row1col1, row1col2, ..., row2col1, ...]
    /// values_per_row: number of columns per row
    /// batch_size: rows per INSERT statement (default 50, matching INSERT_BATCH_SIZE)
    #[napi]
    pub fn bulk_insert(
        &self,
        sql: String,
        flat_values: Vec<napi::JsUnknown>,
        values_per_row: u32,
        batch_size: Option<u32>,
    ) -> napi::Result<u32> {
        let batch_size = batch_size.unwrap_or(50) as usize;
        let vpr = values_per_row as usize;
        let values_per_batch = vpr * batch_size;
        let total_values = flat_values.len();
        let total_rows = total_values / vpr;

        let conn = self.conn.borrow();
        let conn = conn.as_ref().ok_or(...)?;

        let mut batch_stmt = conn.prepare_cached(&sql)?;

        // Build single-row INSERT SQL for remainder
        // (derive from batch SQL by taking first VALUES clause)
        let single_sql = /* extract single-row INSERT from sql */;
        let mut single_stmt = conn.prepare_cached(&single_sql)?;

        let mut inserted = 0u32;
        let mut offset = 0;

        // Full batches
        while offset + values_per_batch <= total_values {
            let params: Vec<rusqlite::types::Value> = flat_values[offset..offset+values_per_batch]
                .iter()
                .map(|v| napi_value_to_sqlite_param(v))
                .collect::<Result<_>>()?;
            batch_stmt.execute(rusqlite::params_from_iter(params))?;
            inserted += batch_size as u32;
            offset += values_per_batch;
        }

        // Remainder rows (one at a time)
        while offset + vpr <= total_values {
            let params: Vec<rusqlite::types::Value> = flat_values[offset..offset+vpr]
                .iter()
                .map(|v| napi_value_to_sqlite_param(v))
                .collect::<Result<_>>()?;
            single_stmt.execute(rusqlite::params_from_iter(params))?;
            inserted += 1;
            offset += vpr;
        }

        Ok(inserted)
    }
}
```

Key implementation details:

- Use `prepare_cached` for both batch and single statements
- Convert napi values to rusqlite params internally (reuse `napi_value_to_sqlite_param` from types.rs)
- Return total rows inserted for verification
- The single-row SQL derivation: find first `)` after `VALUES (` and truncate there, OR accept a second sql parameter

```typescript
function flush() {
  const start = performance.now();
  const flushedRows = pendingRows;
  const flushedSize = pendingSize;

  const flushedValues = flushedRows * valuesPerRow;
  // Single napi crossing for all rows
  db.bulkInsert(
    insertBatchSQL, // the batch INSERT SQL string
    pendingValues.slice(0, flushedValues) as LiteValueType[],
    valuesPerRow,
    INSERT_BATCH_SIZE,
  );

  for (let i = 0; i < flushedValues; i++) {
    pendingValues[i] = undefined as unknown as LiteValueType;
  }
  pendingRows = 0;
  pendingSize = 0;
  status.rows += flushedRows;

  const elapsed = performance.now() - start;
  flushTime += elapsed;

}
```
