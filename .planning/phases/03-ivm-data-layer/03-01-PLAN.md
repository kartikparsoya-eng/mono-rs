```rust
pub fn parse_column_types(env: Env, obj: napi::JsObject) -> napi::Result<Vec<(String, ColumnType)>> {
    let names = obj.get_property_names()?;
    let len = names.get_array_length()?;
    let mut result = Vec::with_capacity(len as usize);
    for i in 0..len {
        let key: napi::JsString = names.get_element(i)?;
        let key_str = key.into_utf8()?.into_owned()?;
        let val: napi::JsString = obj.get_named_property(&key_str)?;
        let val_str = val.into_utf8()?.into_owned()?;
        result.push((key_str, ColumnType::from_str(&val_str)));
    }
    Ok(result)
}
```

<action>
Add to types.rs a function that recursively converts a `serde_json::Value` to a napi JS value:

```rust
pub fn serde_json_to_js(env: Env, value: &serde_json::Value) -> napi::Result<napi::sys::napi_value> {
    use serde_json::Value;
    match value {
        Value::Null => {
            let mut result = std::ptr::null_mut();
            unsafe { napi::sys::napi_get_null(env.raw(), &mut result) };
            Ok(result)
        }
        Value::Bool(b) => {
            let mut result = std::ptr::null_mut();
            unsafe { napi::sys::napi_get_boolean(env.raw(), *b, &mut result) };
            Ok(result)
        }
        Value::Number(n) => {
            let mut result = std::ptr::null_mut();
            let f = n.as_f64().unwrap_or(0.0);
            unsafe { napi::sys::napi_create_double(env.raw(), f, &mut result) };
            Ok(result)
        }
        Value::String(s) => {
            let mut result = std::ptr::null_mut();
            unsafe {
                napi::sys::napi_create_string_utf8(
                    env.raw(), s.as_ptr() as *const i8, s.len(), &mut result
                )
            };
            Ok(result)
        }
        Value::Array(arr) => {
            let mut result = std::ptr::null_mut();
            unsafe { napi::sys::napi_create_array_with_length(env.raw(), arr.len(), &mut result) };
            for (i, item) in arr.iter().enumerate() {
                let js_item = serde_json_to_js(env, item)?;
                unsafe { napi::sys::napi_set_element(env.raw(), result, i as u32, js_item) };
            }
            Ok(result)
        }
        Value::Object(map) => {
            let mut result = std::ptr::null_mut();
            unsafe { napi::sys::napi_create_object(env.raw(), &mut result) };
            for (key, val) in map {
                let js_val = serde_json_to_js(env, val)?;
                let key_cstr = std::ffi::CString::new(key.as_str()).unwrap();
                unsafe {
                    napi::sys::napi_set_named_property(env.raw(), result, key_cstr.as_ptr(), js_val)
                };
            }
            Ok(result)
        }
    }
}
```

## Task 4: Implement row_to_typed_js_object

<read_first>

- packages/zqlite-rs/src/types.rs

<action>
Add function that converts a rusqlite Row to a JS object with type-aware conversion:

```rust
pub fn row_to_typed_js_object(
    env: Env,
    row: &rusqlite::Row,
    columns: &[String],
    column_types: &[(String, ColumnType)],
    table_name: &str,
) -> napi::Result<napi::sys::napi_value> {
    let mut obj = std::ptr::null_mut();
    unsafe { napi::sys::napi_create_object(env.raw(), &mut obj) };

    for (i, col_name) in columns.iter().enumerate() {
        // Find column type
        let col_type = column_types.iter()
            .find(|(name, _)| name == col_name)
            .map(|(_, t)| *t)
            .unwrap_or(ColumnType::String);

        let raw_value = row.get_ref(i).map_err(|e| napi::Error::from_reason(e.to_string()))?;

        let js_val = match raw_value {
            rusqlite::types::ValueRef::Null => {
                let mut v = std::ptr::null_mut();
                unsafe { napi::sys::napi_get_null(env.raw(), &mut v) };
                v
            }
            rusqlite::types::ValueRef::Integer(n) => {
                match col_type {
                    ColumnType::Boolean => {
                        let mut v = std::ptr::null_mut();
                        unsafe { napi::sys::napi_get_boolean(env.raw(), n != 0, &mut v) };
                        v
                    }
                    ColumnType::Number => {
                        // Safe integer bounds check: ±2^53
                        const MAX_SAFE: i64 = 9007199254740991;
                        const MIN_SAFE: i64 = -9007199254740991;
                        if n > MAX_SAFE || n < MIN_SAFE {
                            return Err(napi::Error::from_reason(format!(
                                "value {} (in {}.{}) is outside of supported bounds",
                                n, table_name, col_name
                            )));
                        }
                        let mut v = std::ptr::null_mut();
                        unsafe { napi::sys::napi_create_double(env.raw(), n as f64, &mut v) };
                        v
                    }
                    _ => {
                        // For string/json/null column types with integer value, return as number
                        let mut v = std::ptr::null_mut();
                        unsafe { napi::sys::napi_create_double(env.raw(), n as f64, &mut v) };
                        v
                    }
                }
            }
            rusqlite::types::ValueRef::Real(f) => {
                let mut v = std::ptr::null_mut();
                unsafe { napi::sys::napi_create_double(env.raw(), f, &mut v) };
                v
            }
            rusqlite::types::ValueRef::Text(bytes) => {
                let s = std::str::from_utf8(bytes)
                    .map_err(|e| napi::Error::from_reason(e.to_string()))?;
                match col_type {
                    ColumnType::Json => {
                        let parsed: serde_json::Value = serde_json::from_str(s)
                            .map_err(|e| napi::Error::from_reason(format!(
                                "invalid json value for column {}: {}", col_name, e
                            )))?;
                        serde_json_to_js(env, &parsed)?
                    }
                    _ => {
                        let mut v = std::ptr::null_mut();
                        unsafe {
                            napi::sys::napi_create_string_utf8(
                                env.raw(), s.as_ptr() as *const i8, s.len(), &mut v
                            )
                        };
                        v
                    }
                }
            }
            rusqlite::types::ValueRef::Blob(bytes) => {
                let mut v = std::ptr::null_mut();
                let mut data_ptr = std::ptr::null_mut();
                unsafe {
                    napi::sys::napi_create_buffer_copy(
                        env.raw(), bytes.len(), bytes.as_ptr() as *const _, &mut data_ptr, &mut v
                    )
                };
                v
            }
        };

        let key_cstr = std::ffi::CString::new(col_name.as_str()).unwrap();
        unsafe { napi::sys::napi_set_named_property(env.raw(), obj, key_cstr.as_ptr(), js_val) };
    }

    Ok(obj)
}
```

Key behaviors:

- Boolean columns: INTEGER 0/1 → JS true/false, NULL → null
- Number columns: INTEGER checked against ±2^53 bounds, error message format: `"value {v} (in {table}.{column}) is outside of supported bounds"`
- JSON columns: TEXT parsed via serde_json, converted to JS recursively
- String columns: TEXT → JS string
- NULL always → JS null regardless of column type

```rust
#[napi]
pub fn query_all(
    &self,
    env: Env,
    sql: String,
    params: napi::JsObject,  // JS Array
    column_types: napi::JsObject,  // {colName: typeStr}
    table_name: String,
) -> napi::Result<napi::JsObject> {
    let db = self.db.lock().unwrap();

    // Parse column types once
    let col_types = crate::types::parse_column_types(env, column_types)?;

    // Parse params
    let sqlite_params = crate::types::js_array_params_to_sqlite(env, &params)?;

    // Prepare statement
    let mut stmt = db.prepare(&sql)
        .map_err(|e| napi::Error::from_reason(format!("{}", e)))?;

    // Get column names
    let col_count = stmt.column_count();
    let columns: Vec<String> = (0..col_count)
        .map(|i| stmt.column_name(i).unwrap().to_string())
        .collect();

    // Bind params and execute
    let param_refs: Vec<&dyn rusqlite::types::ToSql> = sqlite_params.iter()
        .map(|v| v as &dyn rusqlite::types::ToSql)
        .collect();

    let mut rows_result = stmt.query(param_refs.as_slice())
        .map_err(|e| napi::Error::from_reason(format!("{}", e)))?;

    // Collect all rows into JS array
    let mut js_rows: Vec<napi::sys::napi_value> = Vec::new();
    while let Some(row) = rows_result.next().map_err(|e| napi::Error::from_reason(format!("{}", e)))? {
        let js_obj = crate::types::row_to_typed_js_object(env, row, &columns, &col_types, &table_name)?;
        js_rows.push(js_obj);
    }

    // Create JS array and fill
    let mut arr = std::ptr::null_mut();
    unsafe { napi::sys::napi_create_array_with_length(env.raw(), js_rows.len(), &mut arr) };
    for (i, val) in js_rows.iter().enumerate() {
        unsafe { napi::sys::napi_set_element(env.raw(), arr, i as u32, *val) };
    }

    // Convert raw napi_value to JsObject
    Ok(unsafe { napi::JsObject::from_raw_unchecked(env.raw(), arr) })
}
```

Note: The `params` parameter is a JsObject (JS Array). Use `js_array_params_to_sqlite` which already handles arrays. The return type is JsObject (which is the JS Array).

Add `use serde_json;` at the top of types.rs if not already present.
</action>

```rust
        assert_eq!(ColumnType::from_str("number"), ColumnType::Number);
        assert_eq!(ColumnType::from_str("string"), ColumnType::String);

        assert_eq!(ColumnType::from_str("unknown"), ColumnType::Null);

    fn test_safe_integer_bounds() {
        const MAX_SAFE: i64 = 9007199254740991;
        const MIN_SAFE: i64 = -9007199254740991;
        // Values within bounds
        assert!(MAX_SAFE <= 9007199254740991);
        assert!(MIN_SAFE >= -9007199254740991);
        // Values outside bounds
        assert!(MAX_SAFE + 1 > 9007199254740991);
        assert!(MIN_SAFE - 1 < -9007199254740991);
    }

    #[test]
    fn test_serde_json_parsing() {
        let val: serde_json::Value = serde_json::from_str(r#"{"a": 1, "b": "hello", "c": [1,2,3]}"#).unwrap();
        assert!(val.is_object());

        let null_val: serde_json::Value = serde_json::from_str("null").unwrap();
        assert!(null_val.is_null());

        let arr_val: serde_json::Value = serde_json::from_str("[1, 2, 3]").unwrap();
        assert!(arr_val.is_array());
    }
}
```
